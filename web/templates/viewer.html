// ── Schema Toggle ──

function toggleSchema() {

  schemaVisible = !schemaVisible;

  document.getElementById('sidebar').classList.toggle('expanded', schemaVisible);

  document.getElementById('toggle-schema').textContent = schemaVisible ? '◀ Collapse' : '▶ Schema';

}

function collapseSchema() {

  schemaVisible = false;

  document.getElementById('sidebar').classList.remove('expanded');

  document.getElementById('toggle-schema').textContent = '▶ Schema';

}

function expandSchema() {

  schemaVisible = true;

  document.getElementById('sidebar').classList.add('expanded');

  document.getElementById('toggle-schema').textContent = '◀ Collapse';

}


// ── Row Highlighting ──

function highlightRow(tableSelector, rowIndex, color) {

  const table = document.querySelector(tableSelector || '.results-table');

  if (!table) return;

  const rows = table.querySelectorAll('tbody tr');

  rows.forEach(r => r.classList.remove('highlighted', 'highlighted-red', 'highlighted-green'));

  if (rows[rowIndex]) {

    rows[rowIndex].classList.add(color === 'red' ? 'highlighted-red' : color === 'green' ? 'highlighted-green' : 'highlighted');

  }

}

function clearHighlights(tableSelector) {

  const table = document.querySelector(tableSelector || '.results-table');

  if (!table) return;

  table.querySelectorAll('tr').forEach(r => r.classList.remove('highlighted', 'highlighted-red', 'highlighted-green'));

}


// ── Cell Callout ──

function annotateCell(tableSelector, rowIndex, colIndex, text) {

  const table = document.querySelector(tableSelector || '.results-table');

  if (!table) return;

  const rows = table.querySelectorAll('tbody tr');

  if (!rows[rowIndex]) return;

  const cells = rows[rowIndex].querySelectorAll('td');

  if (!cells[colIndex]) return;

  cells[colIndex].classList.add('cell-callout');

  cells[colIndex].setAttribute('data-callout', text);

  setTimeout(() => cells[colIndex].classList.add('visible'), 100);

}

function clearAnnotations(tableSelector) {

  const table = document.querySelector(tableSelector || '.results-table');

  if (!table) return;

  table.querySelectorAll('.cell-callout').forEach(c => {

    c.classList.remove('cell-callout', 'visible');

    c.removeAttribute('data-callout');

  });

}

// ── Schema Rendering ──

function renderSchema(tables) {

  currentSchema = tables || [];

  const list = document.getElementById('schema-list');

  list.innerHTML = '';

  document.getElementById('schema-count').textContent = tables.length + ' tables';

  tables.forEach(t => {

    const div = document.createElement('div');

    div.className = 'schema-table';

    div.innerHTML = `

      <div class="schema-table-name" onclick="toggleTable(this)">

        <span class="arrow">▶</span>

        <span>${t.name}</span>

      </div>

      <div class="schema-columns">

        ${t.columns.map(c => `<div class="schema-col">${c.name}<span class="col-type">${c.type}</span></div>`).join('')}

      </div>

    `;

    list.appendChild(div);

  });

}

function toggleTable(el) {

  el.classList.toggle('expanded');

  el.nextElementSibling.classList.toggle('visible');

}

function activateTable(tableName) {

  document.querySelectorAll('.schema-table-name').forEach(el => {

    const nameSpan = el.querySelector('span:last-child');

    if (nameSpan && nameSpan.textContent.trim() === tableName) {

      el.classList.add('expanded');

      el.nextElementSibling.classList.add('visible');

    }

  });

}


// ── showSchema: fetch from API or use cached data ──

function showSchema() {

  if (currentSchema && currentSchema.length > 0) {

    renderSchema(currentSchema);

    return;

  }

  fetch('/api/show-schema', {method: 'POST', headers: {'Content-Type': 'application/json'}})

    .then(r => r.json())

    .then(data => {

      if (data.schema) {

        const tables = [];

        for (const [name, info] of Object.entries(data.schema)) {

          if (!info.is_view) {

            tables.push({

              name: name,

              columns: info.columns || []

            });

          }

        }

        currentSchema = tables;

        renderSchema(tables);

      }

    })

    .catch(err => {

      console.log('Schema API not available (expected for silent renderer):', err);

    });

}

  if (!rows || rows.length === 0) {

    wrap.innerHTML = '<div class="empty-state"><div class="icon">∅</div><span>No results</span></div>';

    document.getElementById('row-count').textContent = '0 rows';

    return;

  }

  let html = '<table class="results-table"><thead><tr>';

  columns.forEach(c => html += `<th>${c}</th>`);

  html += '</tr></thead><tbody>';

  rows.forEach(row => {

    html += '<tr>';

    row.forEach(cell => html += `<td>${cell !== null ? cell : 'NULL'}</td>`);

    html += '</tr>';

  });

  html += '</tbody></table>';

  wrap.innerHTML = html;

  document.getElementById('row-count').textContent = rows.length + ' rows';

  setStatus('Query complete: ' + (queryName || ''), 'ok');

}

function zoomResults() {

  document.getElementById('results-area').classList.add('zoomed');

  document.getElementById('sql-area').style.flex = '0 0 35%';

}

function resetZoom() {

  document.getElementById('results-area').classList.remove('zoomed');

  document.getElementById('sql-area').style.flex = '';

}

function setStatus(text, type) {

  const el = document.getElementById('status-text');

  el.textContent = text;

  el.className = type === 'ok' ? 'status-ok' : type === 'warn' ? 'status-warn' : type === 'err' ? 'status-err' : '';

}

  document.body.style.opacity = '0';

}
